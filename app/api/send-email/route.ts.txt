import { Resend } from 'resend';
import { NextResponse } from 'next/server';

const resend = new Resend(process.env.RESEND_API_KEY);

export async function POST(request: Request) {
  try {
    const body = await request.json();

    const { type, name, phone, email, date, time, guests, message } = body;

    const subject =
      type === 'reservation'
        ? 'Neue Tischreservierung'
        : 'Neue Anfrage über die Webseite';

    const html =
      type === 'reservation'
        ? `
          <h2>Neue Tischreservierung</h2>
          <p><b>Name:</b> ${name}</p>
          <p><b>Telefon:</b> ${phone || '-'}</p>
          <p><b>E-Mail:</b> ${email || '-'}</p>
          <p><b>Datum:</b> ${date}</p>
          <p><b>Uhrzeit:</b> ${time}</p>
          <p><b>Personen:</b> ${guests}</p>
          <p><b>Bemerkung:</b> ${message || '-'}</p>
        `
        : `
          <h2>Neue Anfrage</h2>
          <p><b>Name:</b> ${name}</p>
          <p><b>Telefon:</b> ${phone || '-'}</p>
          <p><b>E-Mail:</b> ${email || '-'}</p>
          <p><b>Nachricht:</b> ${message}</p>
        `;

    await resend.emails.send({
      from: 'Tischreservierung <onboarding@resend.dev>',
      to: process.env.NOTIFY_EMAIL!,
      subject,
      html,
    });

    return NextResponse.json({ success: true });
  } catch (error) {
    return NextResponse.json(
      { success: false, error: 'E-Mail konnte nicht gesendet werden' },
      { status: 500 }
    );
  }
}
